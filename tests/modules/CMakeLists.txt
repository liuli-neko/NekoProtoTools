function(neko_module_test component)
    set(name "module_${component}")
    string(TOUPPER "${component}" macro)
    add_executable(${name} component.cpp component_header.cpp)
    target_compile_definitions(${name} PRIVATE NEKO_TEST_${macro})
    target_link_libraries(${name} PRIVATE NekoProto${component}Module)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
foreach(component IN ITEMS Serializer ArgParser Protocol Transport Rpc JsonRpc Communication)
    if(TARGET NekoProto${component}Module)
        neko_module_test(${component})
    endif()
endforeach()
# Simultaneous imports also exercise shared declarations across sibling modules.
add_executable(module_all component.cpp component_header.cpp)
foreach(component IN ITEMS Serializer ArgParser Protocol Transport Rpc JsonRpc Communication)
    if(TARGET NekoProto${component}Module)
        string(TOUPPER "${component}" macro)
        target_compile_definitions(module_all PRIVATE NEKO_TEST_${macro})
        target_link_libraries(module_all PRIVATE NekoProto${component}Module)
    endif()
endforeach()
add_test(NAME module_all COMMAND module_all)
