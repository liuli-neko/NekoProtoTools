cmake_minimum_required(VERSION 3.28)
project(NekoReflectionModuleConsumer LANGUAGES CXX)
enable_testing()
find_package(NekoProtoTools CONFIG REQUIRED)
add_executable(module_consumer ../../modules/module_main.cpp ../../modules/module_header.cpp)
target_link_libraries(module_consumer PRIVATE NekoProto::ReflectionModule)
target_compile_features(module_consumer PRIVATE cxx_std_20)
set_target_properties(module_consumer PROPERTIES CXX_SCAN_FOR_MODULES ON)
if(NEKO_TEST_NATIVE)
    target_compile_definitions(module_consumer PRIVATE NEKO_TEST_NATIVE)
endif()
add_test(NAME module_consumer COMMAND module_consumer)
add_executable(module_pure ../../modules/module_pure.cpp)
target_link_libraries(module_pure PRIVATE NekoProto::ReflectionModule)
target_compile_features(module_pure PRIVATE cxx_std_20)
set_target_properties(module_pure PROPERTIES CXX_SCAN_FOR_MODULES ON)
add_test(NAME module_pure_import COMMAND module_pure)

foreach(component IN ITEMS Serializer ArgParser Protocol Transport Rpc JsonRpc Communication)
    if(TARGET NekoProto::${component}Module)
        string(TOUPPER "${component}" macro)
        add_executable(module_${component} ../../modules/component.cpp ../../modules/component_header.cpp)
        target_compile_definitions(module_${component} PRIVATE NEKO_TEST_${macro})
        target_link_libraries(module_${component} PRIVATE NekoProto::${component}Module)
        set_target_properties(module_${component} PROPERTIES CXX_SCAN_FOR_MODULES ON)
        add_test(NAME module_${component} COMMAND module_${component})
    endif()
endforeach()
