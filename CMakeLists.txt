cmake_minimum_required(VERSION 3.23)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/version.lua" NEKO_PROTO_VERSION_LINE LIMIT_COUNT 1)
string(REGEX MATCH "[0-9]+[.][0-9]+[.][0-9]+" NEKO_PROTO_VERSION "${NEKO_PROTO_VERSION_LINE}")
if(NOT NEKO_PROTO_VERSION_LINE MATCHES "^function project_version" OR NOT NEKO_PROTO_VERSION)
    message(FATAL_ERROR "version.lua must define project_version() with major.minor.patch")
endif()
project(NekoProtoTools VERSION "${NEKO_PROTO_VERSION}" LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

option(NEKO_PROTO_BUILD_EXAMPLES "Build standalone examples" OFF)
option(NEKO_PROTO_BUILD_TESTS "Build regression and example smoke tests" OFF)
option(NEKO_PROTO_BUILD_MODULES "Build named modules for all enabled components (CMake 3.28+)" OFF)
option(NEKO_PROTO_ENABLE_RAPIDJSON "Enable the RapidJSON serializer" OFF)
option(NEKO_PROTO_ENABLE_YYJSON "Enable the yyjson JSON serializer" OFF)
option(NEKO_PROTO_ENABLE_SIMDJSON "Enable the simdjson JSON serializer" OFF)
option(NEKO_PROTO_ENABLE_PUGIXML "Enable the pugixml XML serializer" OFF)
option(NEKO_PROTO_ENABLE_YAMLCPP "Enable the yaml-cpp YAML serializer" OFF)
option(NEKO_PROTO_ENABLE_LIBFYAML "Enable the libfyaml YAML serializer" OFF)
option(NEKO_PROTO_ENABLE_TOMLPLUSPLUS "Enable the toml++ TOML serializer" OFF)
option(NEKO_PROTO_ENABLE_PROTOCOL "Build the protocol factory" ON)
option(NEKO_PROTO_ENABLE_COMMUNICATION "Build the communication transport" OFF)
option(NEKO_PROTO_ENABLE_JSONRPC "Build JSON-RPC and Neko RPC" OFF)
option(NEKO_PROTO_ENABLE_RPC_TRACE "Enable RPC tracing and monitoring console" OFF)
option(NEKO_PROTO_FETCH_MISSING_DEPS "Fetch missing CMake dependencies during configuration" OFF)

if(NEKO_PROTO_BUILD_TESTS)
    set(NEKO_PROTO_BUILD_EXAMPLES ON)
    enable_testing()
endif()

if(NEKO_PROTO_ENABLE_COMMUNICATION AND NOT NEKO_PROTO_ENABLE_PROTOCOL)
    message(FATAL_ERROR "NEKO_PROTO_ENABLE_COMMUNICATION=ON requires NEKO_PROTO_ENABLE_PROTOCOL=ON.")
endif()
if(NEKO_PROTO_ENABLE_JSONRPC AND NOT (NEKO_PROTO_ENABLE_RAPIDJSON OR NEKO_PROTO_ENABLE_SIMDJSON))
    message(FATAL_ERROR "NEKO_PROTO_ENABLE_JSONRPC=ON requires NEKO_PROTO_ENABLE_RAPIDJSON=ON or NEKO_PROTO_ENABLE_SIMDJSON=ON.")
endif()
if(NEKO_PROTO_ENABLE_RPC_TRACE AND NOT NEKO_PROTO_ENABLE_JSONRPC)
    message(FATAL_ERROR "NEKO_PROTO_ENABLE_RPC_TRACE=ON requires NEKO_PROTO_ENABLE_JSONRPC=ON.")
endif()

if(BUILD_SHARED_LIBS)
    set(NEKO_PROTO_USE_STATIC OFF)
else()
    set(NEKO_PROTO_USE_STATIC ON)
endif()
set(NEKO_PROTO_RPC_TRACE "${NEKO_PROTO_ENABLE_RPC_TRACE}")

include(cmake/NekoProtoToolsExpected.cmake)
if(NOT NEKO_PROTO_EXPECTED_FOUND)
    message(FATAL_ERROR "${NEKO_PROTO_EXPECTED_DIAGNOSTIC}")
endif()
if(NEKO_PROTO_ENABLE_RAPIDJSON)
    include(cmake/NekoProtoToolsRapidJSON.cmake)
    if(NOT NEKO_PROTO_RAPIDJSON_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_RAPIDJSON_DIAGNOSTIC}")
    endif()
endif()
if(NEKO_PROTO_ENABLE_YYJSON)
    include(cmake/NekoProtoToolsYyJSON.cmake)
    if(NOT NEKO_PROTO_YYJSON_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_YYJSON_DIAGNOSTIC}")
    endif()
endif()
if(NEKO_PROTO_ENABLE_SIMDJSON)
    include(cmake/NekoProtoToolsSimdJSON.cmake)
    if(NOT NEKO_PROTO_SIMDJSON_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_SIMDJSON_DIAGNOSTIC}")
    endif()
endif()
if(NEKO_PROTO_ENABLE_PUGIXML)
    include(cmake/NekoProtoToolsPugiXml.cmake)
    if(NOT NEKO_PROTO_PUGIXML_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_PUGIXML_DIAGNOSTIC}")
    endif()
endif()
if(NEKO_PROTO_ENABLE_YAMLCPP)
    include(cmake/NekoProtoToolsYamlCpp.cmake)
    if(NOT NEKO_PROTO_YAMLCPP_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_YAMLCPP_DIAGNOSTIC}")
    endif()
endif()
if(NEKO_PROTO_ENABLE_LIBFYAML)
    include(cmake/NekoProtoToolsLibfyaml.cmake)
    if(NOT NEKO_PROTO_LIBFYAML_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_LIBFYAML_DIAGNOSTIC}")
    endif()
endif()
if(NEKO_PROTO_ENABLE_TOMLPLUSPLUS)
    include(cmake/NekoProtoToolsTomlPlusPlus.cmake)
    if(NOT NEKO_PROTO_TOMLPLUSPLUS_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_TOMLPLUSPLUS_DIAGNOSTIC}")
    endif()
endif()
if(NEKO_PROTO_ENABLE_COMMUNICATION OR NEKO_PROTO_ENABLE_JSONRPC)
    include(cmake/NekoProtoToolsIlias.cmake)
    if(NOT NEKO_PROTO_ILIAS_FOUND)
        message(FATAL_ERROR "${NEKO_PROTO_ILIAS_DIAGNOSTIC}")
    endif()
endif()

configure_file(cmake/config.h.in
    "${CMAKE_CURRENT_BINARY_DIR}/generated/nekoproto/global/config.h" @ONLY)

include(cmake/NekoProtoToolsReflection.cmake)

add_library(NekoProtoSerializer INTERFACE)
add_library(NekoProto::Serializer ALIAS NekoProtoSerializer)
set_target_properties(NekoProtoSerializer PROPERTIES EXPORT_NAME Serializer)
target_compile_features(NekoProtoSerializer INTERFACE cxx_std_20)
target_link_libraries(NekoProtoSerializer INTERFACE NekoProtoReflection)
target_include_directories(NekoProtoSerializer INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NEKO_PROTO_ENABLE_RAPIDJSON)
    target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::RapidJSON)
endif()
if(NEKO_PROTO_ENABLE_YYJSON)
    target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::YyJSON)
endif()
if(NEKO_PROTO_ENABLE_SIMDJSON)
    target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::SimdJSON)
endif()
target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::Expected)
if(NEKO_PROTO_ENABLE_PUGIXML)
    target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::PugiXml)
endif()
if(NEKO_PROTO_ENABLE_YAMLCPP)
    target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::YamlCpp)
endif()
if(NEKO_PROTO_ENABLE_LIBFYAML)
    target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::Libfyaml)
endif()
if(NEKO_PROTO_ENABLE_TOMLPLUSPLUS)
    target_link_libraries(NekoProtoSerializer INTERFACE NekoProto::TomlPlusPlus)
endif()

add_library(NekoProtoArgParser INTERFACE)
add_library(NekoProto::ArgParser ALIAS NekoProtoArgParser)
set_target_properties(NekoProtoArgParser PROPERTIES EXPORT_NAME ArgParser)
target_link_libraries(NekoProtoArgParser INTERFACE NekoProtoSerializer)

if(NEKO_PROTO_ENABLE_PROTOCOL)
    add_library(NekoProtoProtocol src/proto_base.cpp)
    add_library(NekoProto::Protocol ALIAS NekoProtoProtocol)
    set_target_properties(NekoProtoProtocol PROPERTIES EXPORT_NAME Protocol)
    target_link_libraries(NekoProtoProtocol PUBLIC NekoProtoSerializer)
    if(BUILD_SHARED_LIBS)
        target_compile_definitions(NekoProtoProtocol PRIVATE NEKO_PROTO_LIBRARY)
    endif()
endif()

if(NEKO_PROTO_ENABLE_COMMUNICATION)
    add_library(NekoProtoCommunication src/communication_base.cpp)
    add_library(NekoProto::Communication ALIAS NekoProtoCommunication)
    set_target_properties(NekoProtoCommunication PROPERTIES EXPORT_NAME Communication)
    target_link_libraries(NekoProtoCommunication PUBLIC NekoProtoProtocol NekoProto::Ilias)
    if(BUILD_SHARED_LIBS)
        target_compile_definitions(NekoProtoCommunication PRIVATE NEKO_PROTO_LIBRARY)
    endif()
endif()

if(NEKO_PROTO_ENABLE_JSONRPC)
    add_library(NekoProtoJsonRpc src/jsonrpc.cpp src/rpc.cpp)
    add_library(NekoProto::JsonRpc ALIAS NekoProtoJsonRpc)
    set_target_properties(NekoProtoJsonRpc PROPERTIES EXPORT_NAME JsonRpc)
    target_link_libraries(NekoProtoJsonRpc PUBLIC NekoProtoSerializer NekoProto::Ilias)
    if(NEKO_PROTO_ENABLE_RPC_TRACE)
        target_sources(NekoProtoJsonRpc PRIVATE src/rpc_tracing.cpp)
    endif()
    if(BUILD_SHARED_LIBS)
        target_compile_definitions(NekoProtoJsonRpc PRIVATE NEKO_PROTO_LIBRARY)
    endif()
endif()

include(cmake/NekoProtoToolsModules.cmake)

if(NEKO_PROTO_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
if(NEKO_PROTO_BUILD_TESTS)
    add_executable(neko_provider_contract_test tests/unit/reflect/test_provider_contract.cpp)
    target_link_libraries(neko_provider_contract_test PRIVATE NekoProtoReflection)
    add_test(NAME reflection_provider_contract COMMAND neko_provider_contract_test)
    add_subdirectory(tests/cpp26)
    if(NEKO_PROTO_BUILD_MODULES)
        add_subdirectory(tests/modules)
    endif()
endif()
install(DIRECTORY include/nekoproto DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
    FILES_MATCHING PATTERN "*.hpp" PATTERN "*.h" PATTERN "*.h.in" EXCLUDE)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/generated/nekoproto/global/config.h"
    DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/nekoproto/global")

set(_neko_targets NekoProtoReflection NekoProtoSerializer NekoProtoArgParser)
if(NEKO_PROTO_ENABLE_PROTOCOL)
    list(APPEND _neko_targets NekoProtoProtocol)
endif()
if(NEKO_PROTO_ENABLE_COMMUNICATION)
    list(APPEND _neko_targets NekoProtoCommunication)
endif()
if(NEKO_PROTO_ENABLE_JSONRPC)
    list(APPEND _neko_targets NekoProtoJsonRpc)
endif()
install(TARGETS ${_neko_targets} EXPORT NekoProtoToolsTargets
    ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
    LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
    RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
set(_neko_module_export_options)
if(NEKO_PROTO_BUILD_MODULES)
    list(APPEND _neko_module_export_options CXX_MODULES_DIRECTORY cxx-modules)
endif()
install(EXPORT NekoProtoToolsTargets NAMESPACE NekoProto::
    ${_neko_module_export_options}
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/NekoProtoTools")

    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/NekoProtoToolsYamlCpp.cmake"
